class Replay:
 def __init__(self,binary,label):
  self.err=(O/(label+'.stderr')).open('wb');self.p=subprocess.Popen([str(binary),str(NET)],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=self.err,text=True,bufsize=1);self.metadata=json.loads(self.p.stdout.readline(32*1024*1024))
 def request(self,p):
  if p['op']=='load':
   lines=[f"LOAD {len(p['traces'])} {p['branches']} {p['salt']}"]
   for t in p['traces']:lines +=[str(int(t.get('chess960',False))),t['fen'],' '.join(t['moves'])]
  else:lines=[f"RUN {p['workload']} {p['passes']} {p['offset']}"]
  self.p.stdin.write('\n'.join(lines)+'\n');self.p.stdin.flush();return SimpleNamespace(output=json.loads(self.p.stdout.readline(32*1024*1024)))
 def close(self):self.p.kill();self.p.wait();self.err.close()

def calibrate(binary,minimum_depth):
 probes=[]
 for depth in range(minimum_depth,18,2):

   r.request({'op':'search_load','cases':cases,'depth':depth,'hash_mb':16});r.request({'op':'search','passes':1,'offset':0});seconds=r.request({'op':'search','passes':1,'offset':0}).elapsed_seconds
